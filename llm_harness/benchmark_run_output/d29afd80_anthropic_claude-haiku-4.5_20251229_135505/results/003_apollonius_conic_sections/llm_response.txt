// Apollonius's Conic Sections - Ancient Greek Geometric Visualization
// Historical rendering of ellipse, parabola, and hyperbola from a double cone
// Following the mathematical treatise "Conics" (c. 200 BCE)

@vertex
fn vs_main(@builtin(vertex_index) vertex_index: u32) -> @builtin(position) vec4<f32> {
    let vertex_id = vertex_index % 3u;
    let x = f32(i32(vertex_id & 1u) << 2u) - 1.0;
    let y = f32(i32((vertex_id >> 1u) & 1u) << 2u) - 1.0;
    return vec4<f32>(x, y, 0.0, 1.0);
}

struct Params {
    resolution: vec2<f32>,
};

@group(0) @binding(0) var<uniform> params: Params;

// Constants for cone geometry (Apollonius's construction)
const CONE_ANGLE: f32 = 0.523598776f;  // 30 degrees in radians: pi/6
const CONE_HEIGHT: f32 = 3.0;
const PARCHMENT_R: f32 = 0.956862745f;  // #F4E8D0
const PARCHMENT_G: f32 = 0.909803922f;
const PARCHMENT_B: f32 = 0.815686275f;

// Helper: distance from point to cone surface
fn distanceToCone(pos_3d: vec3<f32>) -> f32 {
    let r = length(pos_3d.xy);
    let cone_radius_at_z = abs(pos_3d.z) * tan(CONE_ANGLE);
    return abs(r - cone_radius_at_z);
}

// Helper: determine which conic section is visible at this 3D point
fn conicType(pos_3d: vec3<f32>, plane_type: i32) -> f32 {
    let r = length(pos_3d.xy);
    let z = pos_3d.z;
    let cone_r = abs(z) * tan(CONE_ANGLE);
    
    // Plane equations for each conic
    let ellipse_plane = z * cos(0.785398163f) + r * sin(0.785398163f) - 1.2;  // 45 degrees
    let parabola_plane = z * cos(CONE_ANGLE) + r * sin(CONE_ANGLE) - 1.5;     // 30 degrees (generator)
    let hyperbola_plane = z * cos(0.261799388f) + r * sin(0.261799388f) - 1.0; // 15 degrees
    
    var intersection_dist = 1000.0;
    
    if (plane_type == 0) {
        // Ellipse: intersection distance
        intersection_dist = abs(ellipse_plane);
    } else if (plane_type == 1) {
        // Parabola: intersection distance
        intersection_dist = abs(parabola_plane);
    } else {
        // Hyperbola: intersection distance
        intersection_dist = abs(hyperbola_plane);
    }
    
    return intersection_dist;
}

// Helper: render cone wireframe
fn renderConeWireframe(pos_3d: vec3<f32>) -> f32 {
    let r = length(pos_3d.xy);
    let z = pos_3d.z;
    let cone_r = abs(z) * tan(CONE_ANGLE);
    
    let cone_dist = abs(r - cone_r);
    let wireframe_width = 0.03;
    
    // Generators (vertical lines on cone surface)
    let angle = atan2(pos_3d.y, pos_3d.x);
    let num_generators = 12.0;
    let generator_pattern = fract(angle * num_generators / 6.283185307f);
    let generator_lines = smoothstep(0.05, 0.0, abs(generator_pattern - 0.5));
    
    return smoothstep(wireframe_width, wireframe_width - 0.01, cone_dist) * 0.4 + generator_lines * 0.2;
}

// Helper: project 3D point to 2D screen with orthographic view
fn projectPoint(pos_3d: vec3<f32>, uv: vec2<f32>) -> f32 {
    // Rotate view around x and y axes for isometric-like view
    let angle_x = 0.4;
    let angle_y = 0.6;
    
    let cos_x = cos(angle_x);
    let sin_x = sin(angle_x);
    let cos_y = cos(angle_y);
    let sin_y = sin(angle_y);
    
    // Rotation matrices applied
    let rotated_x = vec3<f32>(
        pos_3d.x,
        pos_3d.y * cos_x - pos_3d.z * sin_x,
        pos_3d.y * sin_x + pos_3d.z * cos_x
    );
    
    let rotated = vec3<f32>(
        rotated_x.x * cos_y + rotated_x.z * sin_y,
        rotated_x.y,
        -rotated_x.x * sin_y + rotated_x.z * cos_y
    );
    
    // Orthographic projection: just use x, y and ignore z depth
    let proj_xy = rotated.xy * 0.35;
    let screen_dist = length(uv - proj_xy);
    
    return screen_dist;
}

// Helper: render conic intersection curve
fn renderConicCurve(uv: vec2<f32>, conic_type: i32) -> vec3<f32> {
    var curve_color = vec3<f32>(0.0, 0.0, 0.0);
    var intensity = 0.0;
    
    // Sweep through 3D space to find conic intersection
    let num_samples = 64.0;
    let samples_z = 6.0;
    
    var min_curve_dist = 1000.0;
    
    for (var iz: i32 = 0; iz < 6; iz = iz + 1) {
        let z_sample = -CONE_HEIGHT + f32(iz) * (2.0 * CONE_HEIGHT) / samples_z;
        let cone_r = abs(z_sample) * tan(CONE_ANGLE);
        
        // Sample around the cone at this z-level
        for (var i: i32 = 0; i < 64; i = i + 1) {
            let angle = f32(i) * 6.283185307f / 64.0;
            let x = cos(angle) * cone_r;
            let y = sin(angle) * cone_r;
            
            let pos_3d = vec3<f32>(x, y, z_sample);
            let screen_dist = projectPoint(pos_3d, uv);
            
            // Check if point is on the cutting plane
            let plane_dist = conicType(pos_3d, conic_type);
            
            // If point is on both cone and plane, it's part of the conic curve
            if (plane_dist < 0.15 && screen_dist < min_curve_dist) {
                min_curve_dist = screen_dist;
            }
        }
    }
    
    let curve_width = 0.008;
    let curve_alpha = smoothstep(curve_width, 0.0, min_curve_dist);
    
    if (conic_type == 0) {
        // Ellipse: deep blue (celestial)
        curve_color = vec3<f32>(0.1, 0.2, 0.6);
    } else if (conic_type == 1) {
        // Parabola: green (earthly)
        curve_color = vec3<f32>(0.2, 0.5, 0.2);
    } else {
        // Hyperbola: red (infinite)
        curve_color = vec3<f32>(0.7, 0.1, 0.1);
    }
    
    return curve_color * curve_alpha;
}

// Helper: render cutting plane
fn renderCuttingPlane(pos_3d: vec3<f32>, conic_type: i32) -> f32 {
    var plane_alpha = 0.0;
    
    if (conic_type == 0) {
        // Ellipse plane: 45 degrees
        let plane_eq = pos_3d.z * cos(0.785398163f) + length(pos_3d.xy) * sin(0.785398163f) - 1.2;
        plane_alpha = smoothstep(0.05, -0.05, plane_eq) * 0.15;
    } else if (conic_type == 1) {
        // Parabola plane: 30 degrees (generator angle)
        let plane_eq = pos_3d.z * cos(CONE_ANGLE) + length(pos_3d.xy) * sin(CONE_ANGLE) - 1.5;
        plane_alpha = smoothstep(0.05, -0.05, plane_eq) * 0.15;
    } else {
        // Hyperbola plane: 15 degrees
        let plane_eq = pos_3d.z * cos(0.261799388f) + length(pos_3d.xy) * sin(0.261799388f) - 1.0;
        plane_alpha = smoothstep(0.05, -0.05, plane_eq) * 0.15;
    }
    
    return plane_alpha;
}

@fragment
fn fs_main(@builtin(position) pos: vec4<f32>) -> @location(0) vec4<f32> {
    let uv = (pos.xy - params.resolution * 0.5) / params.resolution.y;
    
    // Start with parchment background
    var color = vec3<f32>(PARCHMENT_R, PARCHMENT_G, PARCHMENT_B);
    
    // Add subtle grid/texture to parchment
    let grid_scale = 20.0;
    let grid_x = sin(uv.x * grid_scale * 6.283185307f) * 0.05;
    let grid_y = sin(uv.y * grid_scale * 6.283185307f) * 0.05;
    color = color + vec3<f32>(grid_x * 0.02, grid_y * 0.02, 0.0);
    
    // Render the double cone structure
    var cone_sample_color = vec3<f32>(0.0, 0.0, 0.0);
    var max_cone_intensity = 0.0;
    
    // Sample multiple points in 3D to render cone
    for (var sample: i32 = 0; sample < 32; sample = sample + 1) {
        let z_norm = f32(sample) / 32.0;
        let z = -CONE_HEIGHT + z_norm * 2.0 * CONE_HEIGHT;
        
        let cone_r = abs(z) * tan(CONE_ANGLE);
        let sample_angle = atan2(uv.y, uv.x);
        
        let x = cos(sample_angle) * cone_r;
        let y = sin(sample_angle) * cone_r;
        let pos_3d = vec3<f32>(x, y, z);
        
        let proj_dist = projectPoint(pos_3d, uv);
        let cone_intensity = smoothstep(0.02, -0.01, proj_dist) * (1.0 - z_norm * 0.5);
        
        if (cone_intensity > max_cone_intensity) {
            max_cone_intensity = cone_intensity;
            cone_sample_color = vec3<f32>(0.3, 0.25, 0.2) * cone_intensity;
        }
    }
    
    // Blend cone with background
    color = mix(color, cone_sample_color, max_cone_intensity * 0.6);
    
    // Render all three conic sections
    let ellipse_curve = renderConicCurve(uv, 0);
    let parabola_curve = renderConicCurve(uv, 1);
    let hyperbola_curve = renderConicCurve(uv, 2);
    
    // Composite conic curves
    color = color + ellipse_curve;
    color = color + parabola_curve;
    color = color + hyperbola_curve;
    
    // Add labels (simple text approximation using geometry)
    let label_region_x = uv.x;
    let label_region_y = uv.y;
    
    // Ellipse label region (top left)
    if (label_region_x < -0.6 && label_region_y > 0.7) {
        color = mix(color, vec3<f32>(0.1, 0.2, 0.6), 0.1);
    }
    
    // Parabola label region (top center)
    if (abs(label_region_x) < 0.2 && label_region_y > 0.7) {
        color = mix(color, vec3<f32>(0.2, 0.5, 0.2), 0.1);
    }
    
    // Hyperbola label region (top right)
    if (label_region_x > 0.6 && label_region_y > 0.7) {
        color = mix(color, vec3<f32>(0.7, 0.1, 0.1), 0.1);
    }
    
    // Subtle vignette
    let vignette = 1.0 - length(uv) * 0.3;
    color = color * mix(vec3<f32>(0.8), vec3<f32>(1.0), vignette);
    
    // Ensure we stay within valid f32 range
    color = clamp(color, vec3<f32>(0.0), vec3<f32>(1.0));
    
    return vec4<f32>(color, 1.0);
}

// Helper: get conic curve intersection point
fn getConicPoint(angle: f32, z_sample: f32, conic_type: i32) -> vec3<f32> {
    let cone_r = abs(z_sample) * tan(CONE_ANGLE);
    
    var r_adjust = cone_r;
    
    if (conic_type == 0) {
        // Ellipse: reduce radius as z increases
        r_adjust = cone_r * (1.0 - abs(z_sample) * 0.15);
    } else if (conic_type == 1) {
        // Parabola: gradual deformation
        r_adjust = cone_r * (1.0 + z_sample * z_sample * 0.05);
    } else {
        // Hyperbola: increase radius
        r_adjust = cone_r * (1.0 + abs(z_sample) * 0.2);
    }
    
    let x = cos(angle) * r_adjust;
    let y = sin(angle) * r_adjust;
    
    return vec3<f32>(x, y, z_sample);
}

// Helper: render conic intersection curve

    var max_intensity = 0.0;
    
    let num_z_samples = 12.0;
    let num_angle_samples = 96.0;

    for (var iz: i32 = 0; iz < 12; iz = iz + 1) {
        let z_sample = -CONE_HEIGHT + f32(iz) * (2.0 * CONE_HEIGHT) / num_z_samples;
        
        for (var ia: i32 = 0; ia < 96; ia = ia + 1) {
            let angle = f32(ia) * 6.283185307f / num_angle_samples;
            let pos_3d = getConicPoint(angle, z_sample, conic_type);
            
            let screen_dist = projectPoint(pos_3d, uv);
            
            if (screen_dist < min_curve_dist) {
                min_curve_dist = screen_dist;
            }
        }
    }
    
    let curve_width = 0.01;
    let curve_alpha = smoothstep(curve_width, 0.0, min_curve_dist);
    
    if (conic_type == 0) {
        // Ellipse: deep blue
        curve_color = vec3<f32>(0.1, 0.25, 0.7);
    } else if (conic_type == 1) {
        // Parabola: green
        curve_color = vec3<f32>(0.25, 0.6, 0.25);
    } else {
        // Hyperbola: red
        curve_color = vec3<f32>(0.8, 0.15, 0.15);
    }
    
    return curve_color * curve_alpha;
}

    let texture_y = cos(uv.y * texture_scale * 6.283185307f) * 0.03;
    color = color + vec3<f32>(texture_x * 0.01, texture_y * 0.01, 0.0);
    
    // Render cone wireframe
    var max_cone_intensity = 0.0;
    let cone_samples = 32.0;
    let cone_angles = 16.0;
